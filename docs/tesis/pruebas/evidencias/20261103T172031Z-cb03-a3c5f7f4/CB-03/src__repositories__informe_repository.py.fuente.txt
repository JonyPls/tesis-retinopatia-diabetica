 116:     def version_actual_consistente(informe):
 117:         # Lecturas bloqueantes: los SELECT ordinarios podrían ver un snapshot
 118:         # REPEATABLE READ anterior al SELECT FOR UPDATE del informe principal.
 119:         numeros = [numero for (numero,) in db.session.query(InformeVersion.numero_version)
 120:                    .filter_by(informe_id=informe.id)
 121:                    .order_by(InformeVersion.numero_version).with_for_update().all()]
 122:         if (not numeros or len(numeros) != informe.revision_actual or numeros[0] != 1 or
 123:                 numeros[-1] != informe.revision_actual):
 124:             return False
 125:         inicial = (InformeVersion.query.filter_by(informe_id=informe.id, numero_version=1)
 126:                    .with_for_update().populate_existing().first())

 128:             return False
 129:         ultima = (InformeVersion.query.filter_by(informe_id=informe.id,
 130:                                                  numero_version=informe.revision_actual)
 131:                   .with_for_update().populate_existing().first())
 132:         if ultima is None:
 133:             return False
 134:         return (ultima.datos_estructurados == informe.datos_estructurados and
