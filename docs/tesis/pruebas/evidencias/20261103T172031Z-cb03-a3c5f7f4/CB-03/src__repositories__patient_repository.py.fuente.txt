 286:     @staticmethod
 287:     def buscar_paciente(query_text: str, page: int = 1, per_page: int = 5,
 288:                         *, include_last_visit: bool = True):
 289:         search_pattern = f"%{query_text}%"
 290:         prefix_pattern = f"{query_text}%"
 291:         offset = (page - 1) * per_page
 292: 
 293:         count_query = text("""
 294:             SELECT COUNT(*) AS total
 295:             FROM pacientes p
 296:             WHERE p.estado = 'ACTIVO' AND (
 297:                 p.nro_documento LIKE :search_pattern
 298:                 OR CONCAT(COALESCE(p.nombres, ''), ' ', COALESCE(p.apellidos, '')) LIKE :search_pattern
 299:                 OR CONCAT(COALESCE(p.apellidos, ''), ' ', COALESCE(p.nombres, '')) LIKE :search_pattern
 300:             )
 301:         """)
 302: 
 303:         query = text("""
 304:             SELECT
 305:                 p.id,
 306:                 td.codigo AS tipo_doc_abrev,
 307:                 p.tipo_doc_id,
 308:                 td.codigo AS tipo_doc,
 309:                 p.nro_documento,
 310:                 p.nombres,
 311:                 p.apellidos,
 312:                 MAX(CASE WHEN v.fecha_visita_origen = 'UTC_VERIFICADO' THEN v.fecha_visita END) AS ultima_visita,
 313:                 CASE WHEN COUNT(CASE WHEN v.fecha_visita_origen = 'UTC_VERIFICADO' THEN 1 END) > 0 THEN 'UTC_VERIFICADO' ELSE 'NO_VERIFICADO' END AS ultima_visita_origen,
 314:                 CASE WHEN COUNT(CASE WHEN v.fecha_visita_origen = 'NO_VERIFICADO' THEN 1 END) > 0 THEN 1 ELSE 0 END AS visitas_sin_fecha_verificada
 315:             FROM pacientes p
 316:             LEFT JOIN tipos_documento td ON p.tipo_doc_id = td.id
 317:             LEFT JOIN visitas_clinicas v ON p.id = v.paciente_id AND v.estado <> 'ANULADA'
 318:             WHERE p.estado = 'ACTIVO' AND (
 319:                 p.nro_documento LIKE :search_pattern 
 320:                 OR CONCAT(COALESCE(p.nombres, ''), ' ', COALESCE(p.apellidos, '')) LIKE :search_pattern 
 321:                 OR CONCAT(COALESCE(p.apellidos, ''), ' ', COALESCE(p.nombres, '')) LIKE :search_pattern
 322:             )
 323:             GROUP BY
 324:                 p.id,
 325:                 td.codigo,
 326:                 p.tipo_doc_id,
 327:                 p.nro_documento,
 328:                 p.nombres,
 329:                 p.apellidos
 330:             ORDER BY
 331:                 CASE
 332:                     WHEN p.nro_documento LIKE :prefix_pattern THEN 0
 333:                     WHEN CONCAT(COALESCE(p.nombres, ''), ' ', COALESCE(p.apellidos, '')) LIKE :prefix_pattern THEN 1
 334:                     WHEN CONCAT(COALESCE(p.apellidos, ''), ' ', COALESCE(p.nombres, '')) LIKE :prefix_pattern THEN 2
 335:                     ELSE 3
 336:                 END,
 337:                  CASE WHEN COUNT(CASE WHEN v.fecha_visita_origen = 'UTC_VERIFICADO' THEN 1 END) > 0 THEN 0 ELSE 1 END,
 338:                  MAX(CASE WHEN v.fecha_visita_origen = 'UTC_VERIFICADO' THEN v.fecha_visita END) DESC,
 339:                  p.id DESC
 340:             LIMIT :per_page OFFSET :offset
 341:         """) if include_last_visit else text("""
 342:             SELECT
 343:                 p.id,
 344:                 td.codigo AS tipo_doc_abrev,
 345:                 p.tipo_doc_id,
 346:                 td.codigo AS tipo_doc,
 347:                 p.nro_documento,
 348:                 p.nombres,
 349:                 p.apellidos
 350:             FROM pacientes p
 351:             LEFT JOIN tipos_documento td ON p.tipo_doc_id = td.id
 352:             WHERE p.estado = 'ACTIVO' AND (
 353:                 p.nro_documento LIKE :search_pattern
 354:                 OR CONCAT(COALESCE(p.nombres, ''), ' ', COALESCE(p.apellidos, '')) LIKE :search_pattern
 355:                 OR CONCAT(COALESCE(p.apellidos, ''), ' ', COALESCE(p.nombres, '')) LIKE :search_pattern
 356:             )
 357:             ORDER BY
 358:                 CASE
 359:                     WHEN p.nro_documento LIKE :prefix_pattern THEN 0
 360:                     WHEN CONCAT(COALESCE(p.nombres, ''), ' ', COALESCE(p.apellidos, '')) LIKE :prefix_pattern THEN 1
 361:                     WHEN CONCAT(COALESCE(p.apellidos, ''), ' ', COALESCE(p.nombres, '')) LIKE :prefix_pattern THEN 2
 362:                     ELSE 3
 363:                 END,
 364:                 p.id DESC
 365:             LIMIT :per_page OFFSET :offset
 366:         """)
 367:         params = {
 368:             "search_pattern": search_pattern,
 369:             "prefix_pattern": prefix_pattern,

 374:         result = db.session.execute(query, params)

 376:         return [dict(row._mapping) for row in result], total
