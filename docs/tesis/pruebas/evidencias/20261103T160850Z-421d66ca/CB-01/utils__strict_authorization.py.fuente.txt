 108: def evaluate_rule(user, rule: OperationRule, *, resource_allowed=None):
 109:     """Evaluación exacta y con procedencia: solo grants de roles habilitantes."""
 110:     if user is None or getattr(user, "estado", None) != "ACTIVO":
 111:         return False
 112:     if not rule.role_codes or not (rule.all_of or rule.any_of):
 113:         return False
 114:     if any("*" in code for code in rule.all_of | rule.any_of):
 115:         return False
 116:     if resource_allowed is False or (rule.needs_resource_check and resource_allowed is not True):
 117:         return False
 118: 
 119:     permissions = set()
 120:     has_eligible_role = False
 121:     for assignment in getattr(user, "roles_rel", ()) or ():
 122:         role = getattr(assignment, "rol", None)
 123:         if role is None or getattr(role, "estado", None) != "ACTIVO" or getattr(role, "codigo", None) not in rule.role_codes:
 124:             continue
 125:         has_eligible_role = True
 126:         for grant in getattr(role, "permisos_rel", ()) or ():
 127:             permission = getattr(grant, "permiso", None)
 128:             if permission is not None and getattr(permission, "estado", None) == "ACTIVO":
 129:                 permissions.add(permission.codigo)
 130: 
 131:     return (has_eligible_role and rule.all_of <= permissions
 132:             and (not rule.any_of or bool(rule.any_of & permissions)))
 133:

 135: def evaluate_operation(user, operation, *, resource_allowed=None):
 136:     rule = OPERATION_RULES.get(operation)
 137:     return bool(rule and evaluate_rule(user, rule, resource_allowed=resource_allowed))

 140: def patient_read_scope(user):
 141:     """Alcance por grant activo del rol habilitante, nunca por claims del JWT.
 142: 
 143:     Un rol clínico sin su propio pacientes.ver no amplía el alcance concedido
 144:     por otro rol; ADMINISTRADOR_SISTEMA no participa en estas proyecciones.
 145:     """
 146:     rule = OPERATION_RULES["patients.read"]
 147:     for role_code, scope in ((OFTALMOLOGO, "clinical"),
 148:                              (ENFERMERO, "assistance"),
 149:                              (ASISTENTE, "administrative")):
 150:         if evaluate_rule(user, replace(rule, role_codes=frozenset({role_code}))):
 151:             return scope
 152:     return None

 166: class AuthorizationInputError(Exception):
 167:     """Error público de estructura de petición antes de autorizar su modalidad."""

 174: def strict_authorization_required(operation, *, resource_check=None,
 175:                                   conditional_operation=None, requires_conditional_operation=None,
 176:                                   enabled_config=None):
 177:     """Decorador opt-in; envuelve el handler en token_required y deniega por defecto.
 178: 
 179:     Para reglas con needs_resource_check se exige un verificador real del
 180:     recurso. Solo las rutas activadas explícitamente usan este decorador.
 181:     """
 182:     def decorator(handler):
 183:         @wraps(handler)
 184:         def protected(*args, **kwargs):
 185:             claims = getattr(request, "current_user", None) or {}
 186:             if not claims.get("user_id"):
 187:                 return jsonify({"success": False, "message": "No autorizado."}), 401
 188:             # Opt-in: nunca consultar el recurso si la operación aún no fue activada.
 189:             if enabled_config is not None and current_app.config.get(enabled_config) is not True:
 190:                 return jsonify({"success": False, "message": "No autorizado."}), 403
 191:             rule = OPERATION_RULES.get(operation)
 192:             if rule is None:
 193:                 return jsonify({"success": False, "message": "No autorizado."}), 403
 194:             try:
 195:                 user = getattr(request, "authenticated_user", None)
 196:                 if user is None:
 197:                     return jsonify({"success": False, "message": "No autorizado."}), 401
 198:                 if resource_check is None:
 199:                     allowed = evaluate_rule(user, rule)
 200:                 else:
 201:                     # No consultar recursos si el rol/permiso ya deniega acceso.
 202:                     if not evaluate_rule(user, rule, resource_allowed=True):
 203:                         return jsonify({"success": False, "message": "No autorizado."}), 403
 204:                     allowed = evaluate_rule(
 205:                         user, rule, resource_allowed=resource_check(user, *args, **kwargs)
 206:                     )
 207:                 if allowed and conditional_operation is not None:
 208:                     # El selector solo examina la estructura, nunca recursos.
 209:                     # Ambas políticas completas se evalúan en el mismo usuario
 210:                     # y con grants del rol habilitante, sin unión con otros roles.
 211:                     if requires_conditional_operation is None:
 212:                         allowed = False
 213:                     else:
 214:                         required = requires_conditional_operation()
 215:                         if type(required) is not bool:
 216:                             allowed = False
 217:                         elif required:
 218:                             conditional_rule = OPERATION_RULES.get(conditional_operation)
 219:                             allowed = bool(conditional_rule and evaluate_rule(user, conditional_rule))
 220:             except AuthorizationInputError as error:
 221:                 return jsonify({"success": False, "message": str(error)}), error.status_code
 222:             except Exception:
 223:                 current_app.logger.exception("Error al evaluar autorización estricta")
 224:                 return jsonify({"success": False, "message": "No se pudo procesar la solicitud."}), 500
 225:             if not allowed:
 226:                 return jsonify({"success": False, "message": "No autorizado."}), 403
 227:             return handler(*args, **kwargs)
 228: 
 229:         return token_required(protected, eager_permissions=True)
 230: 
 231:     return decorator

 234: def disabled_authenticated_operation(handler):
 235:     """B1: reservar una ruta de escritura sin consultar el recurso ni llamar al handler.
 236: 
 237:     El bloqueo es temporal hasta disponer de versionado y auditoría clínica.
 238:     token_required verifica JWT, cuenta activa y credenciales solo en Bearer.
 239:     """
 240:     @wraps(handler)
 241:     def denied(*args, **kwargs):
 242:         return jsonify({"success": False, "message": "No autorizado."}), 403
 243: 
 244:     return token_required(denied, eager_permissions=True)
