  40: def token_required(f, *, eager_permissions=False):
  41:     @wraps(f)
  42:     def decorated(*args, **kwargs):
  43:         # Rechazar credenciales en URL aun cuando llegue un Bearer valido.
  44:         if any(key.lower() in ('token', 'access_token', 'jwt', 'id_token') or
  45:                any(JWT_URL_PATTERN.fullmatch(value) for value in values)
  46:                for key, values in request.args.lists()):
  47:             return jsonify({"success": False, "message": GENERIC_UNAUTHORIZED_MESSAGE}), 401
  48:         token = None
  49:         
  50:         # 1. Buscamos el token en los encabezados (Headers) de la petición HTTP
  51:         if 'Authorization' in request.headers:
  52:             auth_header = request.headers['Authorization']
  53:             if auth_header.startswith('Bearer '):
  54:                 token = auth_header.split(" ")[1]
  55:         
  56:         # 2. Sin Bearer, bloquear inmediatamente (nunca leer JWT de la URL).
  57:         if not token:
  58:             return jsonify({
  59:                 "success": False, 
  60:                 "message": "Falta el token de autorización. Acceso denegado."
  61:             }), 401
  62: 
  63:         # 3. Validamos que el token sea auténtico y no haya expirado
  64:         try:
  65:             data = jwt.decode(
  66:                 token,
  67:                 os.getenv('SECRET_KEY'),
  68:                 algorithms=["HS256"],
  69:                 options={"require": ["exp", "user_id"]},
  70:             )
  71:             
  72:         except jwt.ExpiredSignatureError:
  73:             return jsonify({
  74:                 "success": False, 
  75:                 "message": "Token expirado."
  76:             }), 401
  77:         except jwt.InvalidTokenError:
  78:             return jsonify({
  79:                 "success": False, 
  80:                 "message": "Token inválido."
  81:             }), 401
  82: 
  83:         try:
  84:             if eager_permissions:
  85:                 user = AuthRepository.get_active_user_with_permissions_by_id(data['user_id'])
  86:             else:
  87:                 user = AuthRepository.get_active_user_by_id(data['user_id'])
  88:             if not user:
  89:                 return jsonify({"success": False, "message": GENERIC_UNAUTHORIZED_MESSAGE}), 401
  90: 
  91:             # Los roles y permisos del token pueden haber cambiado desde su emisión.
  92:             data.pop('permissions', None)
  93:             request.current_user = {**data, 'roles': user.roles, 'permisos': user.permisos}
  94:             if eager_permissions:
  95:                 request.authenticated_user = user  # Solo durante esta petición.
  96:         except Exception:
  97:             current_app.logger.exception("Error al verificar la cuenta autenticada")
  98:             return safe_json_error(status=500)
  99: 
 100:         # 4. Si todo está perfecto, dejamos que la petición continúe hacia la ruta original
 101:         return f(*args, **kwargs)
 102:         
 103:     return decorated

 110: def _as_set(values):
 111:     if not values:
 112:         return set()
 113:     if isinstance(values, str):
 114:         return {values}
 115:     return {str(value) for value in values if value}

 146: def _permission_matches(user_permission, required_permission):
 147:     if user_permission == required_permission:
 148:         return True
 149:     if user_permission.endswith(".*"):
 150:         return required_permission.startswith(user_permission[:-1])
 151:     if required_permission.endswith(".*"):
 152:         return user_permission.startswith(required_permission[:-1])
 153:     return False

 156: def user_has_role(*roles):
 157:     expected_roles = _as_set(roles)
 158:     user_roles = current_user_roles()
 159:     return bool(user_roles & expected_roles)

 162: def user_has_permission(*permissions):
 163:     expected_permissions = _as_set(permissions)
 164:     user_permissions = current_user_permissions()
 165:     return any(
 166:         _permission_matches(user_permission, expected_permission)
 167:         for user_permission in user_permissions
 168:         for expected_permission in expected_permissions
 169:     )

 172: def user_is_authorized(roles=None, permissions=None):
 173:     expected_roles = _as_set(roles)
 174:     expected_permissions = _as_set(permissions)
 175: 
 176:     if current_user_roles() & ADMIN_ROLES:
 177:         return True
 178:     if expected_roles and user_has_role(*expected_roles):
 179:         return True
 180:     if expected_permissions and user_has_permission(*expected_permissions):
 181:         return True
 182:     return False

 185: def authorization_required(roles=None, permissions=None):
 186:     def decorator(f):
 187:         @wraps(f)
 188:         def wrapped(*args, **kwargs):
 189:             if not user_is_authorized(roles=roles, permissions=permissions):
 190:                 return jsonify({
 191:                     "success": False,
 192:                     "message": GENERIC_UNAUTHORIZED_MESSAGE,
 193:                 }), 403
 194:             return f(*args, **kwargs)
 195: 
 196:         return wrapped
 197: 
 198:     return decorator
