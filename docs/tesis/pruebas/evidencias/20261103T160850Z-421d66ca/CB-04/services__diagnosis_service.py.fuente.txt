 186: def _undo_creation(files, commit_attempted, detalle_id):
 187:     """No borrar archivos si un COMMIT ambiguo pudo haber confirmado su fila."""
 188:     try:
 189:         db.session.rollback()
 190:         rollback_ok = True
 191:     except Exception:
 192:         rollback_ok = False
 193:         current_app.logger.error('Rollback de diagnóstico no confirmado; requiere conciliación')
 194: 
 195:     safe_to_remove = rollback_ok and not commit_attempted
 196:     if commit_attempted and detalle_id is not None:
 197:         try:
 198:             # Conexión independiente: nunca decidir usando la sesión rota.
 199:             with db.engine.connect() as connection:
 200:                 row = connection.execute(text(
 201:                     'SELECT imagen_original_url, imagen_gradcam_url '
 202:                     'FROM diagnosticos_retinopatia_detalle WHERE id = :id'
 203:                 ), {'id': detalle_id}).first()
 204:             safe_to_remove = rollback_ok and row is None
 205:             if row is not None:
 206:                 current_app.logger.warning('Commit de diagnóstico confirmado o ambiguo; conservar archivos')
 207:         except Exception:
 208:             safe_to_remove = False
 209:             current_app.logger.error('Estado del commit no verificable; conservar archivos')
 210:     if files is not None:
 211:         try:
 212:             files.cleanup(published=safe_to_remove)
 213:         except OSError:
 214:             current_app.logger.error('Limpieza incompleta; requiere conciliación')
 215:         finally:
 216:             files.release()

 227: def _consultar_detalles(visita_id: int = None, diagnostico_id: int = None) -> dict:
 228:     try:
 229:         if visita_id is not None:
 230:             detalles = DiagnosisRepository.obtener_detalles_por_visita(visita_id)
 231:             response_id = visita_id
 232:         else:
 233:             detalles = DiagnosisRepository.obtener_detalles_por_diagnostico(diagnostico_id)
 234:             response_id = diagnostico_id
 235: 
 236:         if detalles is None:
 237:             return {"success": False, "message": DIAGNOSIS_NOT_FOUND_MESSAGE}
 238: 
 239:         ojos = []
 240:         for d in detalles:
 241:             ojo = "OD" if d['ojo_evaluado'] == 'Derecho' else "OI"
 242:             resultado_completo = d.get('resultado_completo')
 243:             if isinstance(resultado_completo, str):
 244:                 resultado_completo = json.loads(resultado_completo)
 245: 
 246:             diagnostico_texto = None
 247:             probabilidades_detalle = None
 248:             if resultado_completo:
 249:                 diagnostico_texto = resultado_completo.get('diagnostico_texto')
 250:                 probabilidades_detalle = resultado_completo.get('probabilidades_detalle')
 251:             if not diagnostico_texto:
 252:                 diagnostico_texto = RD_LABELS.get(d['resultado_ia'], "Desconocido")
 253: 
 254:             gradcam_url = d.get('gradcam_url')
 255:             if not gradcam_url and d['imagen_url']:
 256:                 gradcam_url = d['imagen_url'].replace('.jpg', '_gradcam.jpg')
 257: 
 258:             ojo_data = {
 259:                 "detalle_id": d['id'],
 260:                 "ojo": ojo,
 261:                 "resultado_ia": d['resultado_ia'],
 262:                 "diagnostico_texto": diagnostico_texto,
 263:                 "confianza_general": float(d['probabilidad_ia']) * 100.0 if d['probabilidad_ia'] is not None else 0.0,
 264:                 "probabilidades_detalle": probabilidades_detalle,
 265:                 "diagnostico_final": d['diagnostico_final'],
 266:                 "estado_validacion": d['estado_validacion'],
 267:                 "comentario_medico": d['comentario_medico'],
 268:                 "motivo_correccion": d['motivo_correccion'],
 269:                 "imagen_url": d['imagen_url'],
 270:                 "gradcam_url": gradcam_url,
 271:             }
 272:             ojos.append(ojo_data)
 273: 
 274:         return {
 275:             "success": True,
 276:             "data": {
 277:                 "visita_id": response_id if visita_id is not None else None,
 278:                 "diagnostico_id": response_id,
 279:                 "ojos": ojos
 280:             }
 281:         }
 282:     except Exception:
 283:         current_app.logger.error("Error al consultar detalles de diagnostico")
 284:         return {"success": False, "message": "No se pudo procesar la solicitud."}
