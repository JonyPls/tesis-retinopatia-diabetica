   1: import io
   2: import logging
   3: import warnings
   4: import cv2
   5: import numpy as np
   6: from PIL import Image
   7: from transformers import CLIPProcessor, CLIPModel
   8: import torch
   9: import logging

  11: # Capa 1: Validación de formato y dimensiones ------------------------
  12: 
  13: FORMATOS_PERMITIDOS = {"JPEG", "PNG", "TIFF"}
  14: DIMENSION_MINIMA    = 350  # px en el lado menor, basado en análisis del dataset EyePACS
  15: MAX_IMAGE_SIDE = 8192
  16: MAX_IMAGE_PIXELS = 30_000_000
  17: MAX_IMAGE_BYTES = 20 * 1024 * 1024
  18: IMAGE_TYPES = {
  19:     "JPEG": ({".jpg", ".jpeg"}, "image/jpeg"),
  20:     "PNG": ({".png"}, "image/png"),
  21:     "TIFF": ({".tif", ".tiff"}, "image/tiff"),
  22: }
  23: 
  24: def validar_formato_y_dimensiones(image_bytes: bytes, *, filename=None, mimetype=None,
  25:                                   max_side=MAX_IMAGE_SIDE, max_pixels=MAX_IMAGE_PIXELS) -> dict:
  26:     """
  27:     Verifica que el archivo sea una imagen válida, tenga un formato
  28:     permitido y dimensiones mínimas compatibles con una retinografía.
  29: 
  30:     Retorna:
  31:         {"valido": True} si pasa todas las verificaciones.
  32:         {"valido": False, "error": "<motivo>"} si falla alguna.
  33:     """
  34:     # 1. Verificar que los bytes correspondan a una imagen legible
  35:     if not image_bytes:
  36:         return {"valido": False, "error": "El archivo está vacío."}
  37:     if filename is not None:
  38:         if (not filename or '/' in filename or '\\' in filename or
  39:                 filename in ('.', '..') or len(filename) > 255 or
  40:                 any(ord(char) < 32 for char in filename)):
  41:             return {"valido": False, "error": "Nombre de imagen inválido."}
  42:     try:
  43:         with warnings.catch_warnings():
  44:             warnings.simplefilter('error', Image.DecompressionBombWarning)
  45:             with Image.open(io.BytesIO(image_bytes)) as imagen:
  46:                 formato = imagen.format
  47:                 ancho, alto = imagen.size
  48:                 if formato not in FORMATOS_PERMITIDOS:
  49:                     return {"valido": False, "error": "Formato de imagen no permitido."}
  50:                 if min(ancho, alto) < DIMENSION_MINIMA:
  51:                     return {"valido": False, "error": "Resolución insuficiente."}
  52:                 if max(ancho, alto) > max_side or ancho * alto > max_pixels:
  53:                     return {"valido": False, "error": "Dimensiones de imagen excedidas."}
  54:                 if filename is not None:
  55:                     extension = '.' + filename.rsplit('.', 1)[-1].lower()
  56:                     if extension not in IMAGE_TYPES[formato][0] or mimetype != IMAGE_TYPES[formato][1]:
  57:                         return {"valido": False, "error": "Extensión o tipo MIME incompatible con la imagen."}
  58:                 imagen.verify()
  59:             # verify comprueba estructura; load comprueba también los píxeles.
  60:             with Image.open(io.BytesIO(image_bytes)) as imagen:
  61:                 imagen.load()
  62:     except Exception:
  63:         return {
  64:             "valido": False,
  65:             "error": "El archivo no es una imagen válida o está corrupto."
  66:         }
  67: 
  68:     return {"valido": True}
  69:

  71: # Capa 2: Validación de región circular dominante ------------------------
  72: 
  73: UMBRAL_AREA_CIRCULAR = 0.40  # La región circular debe ocupar al menos ese porcentaje del área total
  74: 
  75: def validar_region_circular(image_bytes: bytes) -> dict:
  76:     """
  77:     Verifica que la imagen contenga una región circular dominante
  78:     sobre fondo oscuro, característica principal de una retinografía.
  79: 
  80:     Retorna:
  81:         {"valido": True} si se detecta una región circular suficiente.
  82:         {"valido": False, "error": "<motivo>"} si no se detecta.
  83:     """
  84:     # 1. Decodificar bytes a matriz OpenCV
  85:     nparr   = np.frombuffer(image_bytes, np.uint8)
  86:     img_bgr = cv2.imdecode(nparr, cv2.IMREAD_COLOR)
  87: 
  88:     if img_bgr is None:
  89:         return {
  90:             "valido": False,
  91:             "error": "No se pudo decodificar la imagen para análisis."
  92:         }
  93: 
  94:     area_total = img_bgr.shape[0] * img_bgr.shape[1]
  95: 
  96:     # 2. Mismo pipeline de detección de contornos que inference.py
  97:     gray     = cv2.cvtColor(img_bgr, cv2.COLOR_BGR2GRAY)
  98:     umbral   = max(10, int(gray.max() * 0.05))
  99:     blur     = cv2.GaussianBlur(gray, (5, 5), 0)
 100:     _, mask  = cv2.threshold(blur, umbral, 255, cv2.THRESH_BINARY)
 101:     mask     = cv2.morphologyEx(mask, cv2.MORPH_CLOSE, np.ones((11, 11), np.uint8))
 102: 
 103:     contornos, _ = cv2.findContours(mask, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
 104: 
 105:     # 3. Si no hay contornos, la imagen no tiene ninguna región diferenciable
 106:     if not contornos:
 107:         return {
 108:             "valido": False,
 109:             "error": "No se detectó ninguna región de interés en la imagen. "
 110:                      "Verifique que sea una retinografía de fondo de ojo."
 111:         }
 112: 
 113:     # 4. Evaluar el contorno principal (el de mayor área)
 114:     contorno_principal = max(contornos, key=cv2.contourArea)
 115:     area_contorno      = cv2.contourArea(contorno_principal)
 116:     proporcion         = area_contorno / area_total
 117: 
 118:     if proporcion < UMBRAL_AREA_CIRCULAR:
 119:         return {
 120:             "valido": False,
 121:             "error": "La imagen no presenta una región retiniana dominante. "
 122:                      "Verifique que sea una retinografía de fondo de ojo."
 123:         }
 124: 
 125:     return {"valido": True}

 128: # Capa 3: Validación semántica con CLIP ------------------------
 129: 
 130: CLIP_MODEL_ID = "openai/clip-vit-base-patch32"
 131: 
 132: CLIP_PROMPTS = [
 133:     "a retinal fundus photograph of an eye",            # [0] → aceptar
 134:     "a photograph of a planet or astronomical object",  # [1] → rechazar
 135:     "a photograph of an animal or pet",                 # [2] → rechazar
 136:     "a photograph of a person or face",                 # [3] → rechazar
 137:     "a document, screenshot or text image",             # [4] → rechazar
 138:     "a landscape, nature or outdoor photograph",        # [5] → rechazar
 139:     "a generic or unrecognized image",                  # [6] → rechazar
 140: ]
 141: 
 142: PROMPT_VALIDO_IDX = 0
 143: 
 144: MENSAJES_RECHAZO = {
 145:     1: "Se detectó una imagen astronómica. Por favor suba una retinografía de fondo de ojo.",
 146:     2: "Se detectó la fotografía de un animal. Por favor suba una retinografía de fondo de ojo.",
 147:     3: "Se detectó la fotografía de una persona. Por favor suba una retinografía de fondo de ojo.",
 148:     4: "Se detectó un documento o captura de pantalla. Por favor suba una retinografía de fondo de ojo.",
 149:     5: "Se detectó una fotografía de paisaje. Por favor suba una retinografía de fondo de ojo.",
 150:     6: "La imagen no fue reconocida como una retinografía. Por favor suba una retinografía de fondo de ojo.",
 151: }
 152: 
 153: _clip_model     = None
 154: _clip_processor = None
 155: 
 156: def _get_clip():
 157:     global _clip_model, _clip_processor
 158:     if _clip_model is None:
 159:         _clip_model     = CLIPModel.from_pretrained(CLIP_MODEL_ID)
 160:         _clip_processor = CLIPProcessor.from_pretrained(CLIP_MODEL_ID)
 161:         _clip_model.eval()
 162:     return _clip_model, _clip_processor
 163: 
 164: 
 165: def validar_contenido_semantico(image_bytes: bytes) -> dict:
 166:     try:
 167:         modelo, processor = _get_clip()
 168:     except Exception:
 169:         logging.warning("CLIP no disponible para validar la imagen")
 170:         return {"valido": False, "error": "Validación de imagen no disponible.", "status": 503}
 171: 
 172:     try:
 173:         imagen = Image.open(io.BytesIO(image_bytes)).convert("RGB")
 174:     except Exception:
 175:         logging.error("Error interno al preparar imagen para CLIP")
 176:         return {"valido": False, "error": "No se pudo procesar la solicitud.", "status": 500}
 177: 
 178:     try:
 179:         inputs = processor(
 180:             text=CLIP_PROMPTS,
 181:             images=imagen,
 182:             return_tensors="pt",
 183:             padding=True
 184:         )
 185: 
 186:         with torch.no_grad():
 187:             outputs = modelo(**inputs)
 188:             logits  = outputs.logits_per_image
 189:             probs   = logits.softmax(dim=1).squeeze()
 190:     except Exception:
 191:         logging.warning("CLIP no disponible para validar la imagen")
 192:         return {"valido": False, "error": "Validación de imagen no disponible.", "status": 503}
 193:     try:
 194:         idx_ganador = int(probs.argmax())
 195:     except Exception:
 196:         logging.error("CLIP devolvió un resultado inválido")
 197:         return {"valido": False, "error": "No se pudo procesar la solicitud.", "status": 500}
 198:     if idx_ganador != PROMPT_VALIDO_IDX:
 199:         mensaje = MENSAJES_RECHAZO.get(
 200:             idx_ganador,
 201:             "La imagen no corresponde a una retinografía de fondo de ojo."
 202:         )
 203:         return {"valido": False, "error": mensaje}
 204:     return {"valido": True}
