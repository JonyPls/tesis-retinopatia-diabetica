  34: def _clean_text(value):
  35:     if value is None:
  36:         return None
  37:     normalized = ' '.join(str(value).strip().split())

  41: def _valid_person_name(value):
  42:     if not value or any(unicodedata.category(char) in ('Cc', 'Cf') for char in value):
  43:         return False
  44:     name = unicodedata.normalize('NFC', value)
  45:     tokens = re.split(r" ?[-'’] ?| ", name)
  46:     return bool(tokens) and all(token and all(
  47:         char.isalpha() and unicodedata.name(char, '').startswith('LATIN ')
  48:         for char in token) for token in tokens)

  58: def _normalize_document(value, tipo_doc):
  59:     raw = str(value or '').strip()
  60:     if tipo_doc == 'DNI':
  61:         return re.sub(r'\D', '', raw)
  62:     return re.sub(r'[^A-Za-z0-9]', '', raw).upper()

  65: def _parse_birth_date(value, *, today=None):
  66:     try:
  67:         if not isinstance(value, str) or not re.fullmatch(r'\d{4}-\d{2}-\d{2}', value):
  68:             raise ValueError('invalid civil date')
  69:         parsed = date.fromisoformat(value)
  70:     except (TypeError, ValueError):
  71:         return None, 'Verifica la fecha de nacimiento ingresada.'
  72: 
  73:     today = today or datetime.now(CLINIC_ZONE).date()
  74:     if parsed < MIN_BIRTH_DATE:
  75:         return None, 'La fecha de nacimiento debe ser igual o posterior al 01/01/1900.'
  76:     if parsed > today:
  77:         return None, 'La fecha de nacimiento no puede ser futura.'
  78: 
  79:     age = today.year - parsed.year - (
  80:         (today.month, today.day) < (parsed.month, parsed.day)
  81:     )
  82:     if age > MAX_PATIENT_AGE:
  83:         return None, f'La edad del paciente no puede superar {MAX_PATIENT_AGE} años.'
  84: 
  85:     return parsed, None

  88: def _validate_patient_data(
  89:     tipo_doc_codigo,
  90:     nro_documento,
  91:     nombres,
  92:     apellidos,
  93:     sexo,
  94:     fecha_nacimiento,
  95:     telefono,
  96:     correo,
  97:     consentimiento_informado,
  98: ):
  99:     errors = {}
 100: 
 101:     pattern_config = DOCUMENT_PATTERNS.get(tipo_doc_codigo)
 102:     if not pattern_config:
 103:         errors['tipo_doc'] = 'Tipo de documento inválido.'
 104:     elif not pattern_config[0].fullmatch(nro_documento):
 105:         errors['nro_documento'] = pattern_config[1]
 106: 
 107:     if not nombres:
 108:         errors['nombres'] = 'Debe ingresar nombres.'
 109:     elif len(nombres) < 2 or len(nombres) > 80:
 110:         errors['nombres'] = 'Los nombres deben tener entre 2 y 80 caracteres.'
 111:     elif not _valid_person_name(nombres):
 112:         errors['nombres'] = 'Ingresa un nombre válido, sin números ni símbolos.'
 113:     if not apellidos:
 114:         errors['apellidos'] = 'Debe ingresar apellidos.'
 115:     elif len(apellidos) < 2 or len(apellidos) > 100:
 116:         errors['apellidos'] = 'Los apellidos deben tener entre 2 y 100 caracteres.'
 117:     elif not _valid_person_name(apellidos):
 118:         errors['apellidos'] = 'Ingresa un apellido válido, sin números ni símbolos.'
 119: 
 120:     if sexo not in ('Masculino', 'Femenino'):
 121:         errors['sexo'] = 'Debe seleccionar un sexo válido.'
 122: 
 123:     if telefono and not re.match(r'^9\d{8}$', telefono):
 124:         errors['telefono'] = 'El teléfono debe tener 9 dígitos y empezar con 9.'
 125: 
 126:     if correo and (len(correo) > 150 or not _valid_email(correo)):
 127:         errors['correo'] = 'Ingresa un correo electrónico válido.'
 128: 
 129:     if not consentimiento_informado:
 130:         errors['consent'] = 'Debe registrar el consentimiento informado del paciente.'
 131: 
 132:     return errors

 146: def registrar_nuevo_paciente(datos_paciente_dict, *, create_scope, read_scope):
 147:     allowed = PATIENT_CREATE_FIELDS.get(create_scope)

 151:         return {"success": False, "message": "La solicitud contiene campos no permitidos."}, 400
 152:     if any(not isinstance(value, (str, type(None))) for key, value in
 153:            datos_paciente_dict.items() if key != 'consentimiento_informado'):
 154:         return {"success": False, "message": "La solicitud contiene valores no válidos."}, 400
 155:     if ('consentimiento_informado' in datos_paciente_dict and

 157:         return {"success": False, "message": "La solicitud contiene valores no válidos."}, 400

 191:         correo,
 192:         consentimiento_informado,
